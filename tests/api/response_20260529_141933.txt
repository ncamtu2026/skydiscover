def search_algorithm(src, dsts, G, num_partitions):
    bc = BroadCastTopology(src, dsts, num_partitions)
    covered = {src}
    remaining = set(dsts)
    # Greedy Steiner tree: iteratively add the closest uncovered destination
    while remaining:
        best_dst = None
        best_path = None
        best_cost = float('inf')
        for u in covered:
            # Dijkstra from u to all nodes
            try:
                distances, predecessors = nx.single_source_dijkstra(G, u, weight='cost')
            except nx.NetworkXNoPath:
                continue
            for v in remaining:
                if v in distances and distances[v] < best_cost:
                    best_cost = distances[v]
                    best_dst = v
                    # reconstruct path (list of edges)
                    path_edges = []
                    node = v
                    while node != u:
                        pred = predecessors[node]
                        path_edges.append((pred, node))
                        node = pred
                    path_edges.reverse()
                    best_path = path_edges
        # add edges of best_path to tree (only for nodes not already covered)
        for s, t in best_path:
            if t not in covered:
                covered.add(t)
                for j in range(num_partitions):
                    bc.append_dst_partition_path(t, j, [s, t, G[s][t]])
        remaining.remove(best_dst)
    return bc
